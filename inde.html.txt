      const randomTime = Math.floor(Math.random() * (60000 - 20000 + 1)) + 20000;
      
      inactivityTimer = setTimeout(() => {
        triggerInactivityMessage();
      }, randomTime);
    }

    function triggerInactivityMessage() {
      showTypingIndicator();
      setTimeout(() => {
        removeTypingIndicator();
        const randomMsg = inactivityMessages[Math.floor(Math.random() * inactivityMessages.length)];
        addMessage(randomMsg, 'received');
        resetInactivityTimer();
      }, 2000);
    }

    // Adaptar input
    messageInput.addEventListener('input', () => {
      resetInactivityTimer();
      if (messageInput.value.trim().length > 0) {
        sendBtn.style.display = 'block';
        heartBtn.style.display = 'none';
      } else {
        sendBtn.style.display = 'none';
        heartBtn.style.display = 'flex';
      }
    });

    function sendMessage() {
      const text = messageInput.value.trim();
      if (!text) return;

      addMessage(text, 'sent');
      messageInput.value = '';
      sendBtn.style.display = 'none';
      heartBtn.style.display = 'flex';

      resetInactivityTimer();

      headerStatus.innerText = 'Escribiendo...';
      
      setTimeout(() => {
        showTypingIndicator();
      }, 500);

      const delay = Math.floor(Math.random() * 1000) + 1200;
      setTimeout(() => {
        removeTypingIndicator();
        headerStatus.innerText = 'Activa(o) ahora';
        generateAmoeResponse(text);
        resetInactivityTimer();
      }, delay);
    }

    function sendQuickMsg(text) {
      messageInput.value = text;
      sendMessage();
    }

    sendBtn.addEventListener('click', sendMessage);
    messageInput.addEventListener('keypress', (e) => {
      if (e.key === 'Enter') sendMessage();
    });

    function addMessage(text, type) {
      const msgDiv = document.createElement('div');
      msgDiv.classList.add('message', type);

      const timeStr = new Date().toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' });

      msgDiv.innerHTML = `
        <div class="message-content">
          ${escapeHTML(text)}
          <div class="message-time">${timeStr}</div>
        </div>
      `;

      chatArea.appendChild(msgDiv);
      scrollToBottom();
    }

    function showTypingIndicator() {
      if (document.getElementById('typingIndicator')) return;
      const typingDiv = document.createElement('div');
      typingDiv.id = 'typingIndicator';
      typingDiv.classList.add('message', 'received');
      typingDiv.innerHTML = `
        <div class="typing-indicator">
          <div class="typing-dot"></div>
          <div class="typing-dot"></div>
          <div class="typing-dot"></div>
        </div>
      `;
      chatArea.appendChild(typingDiv);
      scrollToBottom();
    }

    function removeTypingIndicator() {
      const typingDiv = document.getElementById('typingIndicator');
      if (typingDiv) typingDiv.remove();
    }

    function escapeHTML(str) {
      return str.replace(/[&<>'"]/g, 
        tag => ({
          '&': '&amp;',
          '<': '&lt;',
          '>': '&gt;',
          "'": '&#39;',
          '"': '&quot;'
        }[tag] || tag)
      );
    }
